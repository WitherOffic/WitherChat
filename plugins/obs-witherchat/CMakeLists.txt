cmake_minimum_required(VERSION 3.28)
project(witherchat_obs VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The first dock prototype supports Windows x64/MSVC only.")
endif()
set(OBS_SOURCE_DIR "" CACHE PATH "OBS Studio 32.2.2 source tree")
if(NOT EXISTS "${OBS_SOURCE_DIR}/libobs/obs-module.h")
  message(FATAL_ERROR "Set OBS_SOURCE_DIR to the official OBS 32.2.2 source tree.")
endif()
find_package(Qt6 6.11.1 EXACT COMPONENTS Core Gui Widgets REQUIRED)
# Compile against official headers; resolve the frontend ABI from OBS's loaded DLLs.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/obsconfig.h"
  "#pragma once\n#define OBS_INSTALL_PREFIX \"\"\n#define OBS_DATA_PATH \"data/obs-studio\"\n")
add_library(witherchat-obs MODULE src/plugin.cpp)
target_compile_features(witherchat-obs PRIVATE cxx_std_17)
target_compile_definitions(witherchat-obs PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(witherchat-obs SYSTEM PRIVATE
  "${OBS_SOURCE_DIR}/libobs" "${OBS_SOURCE_DIR}/frontend/api")
target_include_directories(witherchat-obs PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(witherchat-obs PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets user32)
target_compile_options(witherchat-obs PRIVATE /W4 /WX /utf-8 /permissive-)
option(WITHERCHAT_OBS_DOCK_DIAGNOSTICS "Enable isolated automated native dock probe" OFF)
if(WITHERCHAT_OBS_DOCK_DIAGNOSTICS)
  target_compile_definitions(witherchat-obs PRIVATE WITHERCHAT_OBS_DOCK_DIAGNOSTICS)
endif()
option(WITHERCHAT_OBS_DOCK_TESTS "Build isolated native helper regression tests" OFF)
if(WITHERCHAT_OBS_DOCK_TESTS)
  enable_testing()
  add_executable(witherchat-obs-tests tests/worker-support-tests.cpp)
  target_compile_features(witherchat-obs-tests PRIVATE cxx_std_17)
  target_include_directories(witherchat-obs-tests PRIVATE src)
  target_link_libraries(witherchat-obs-tests PRIVATE Qt6::Core)
  target_compile_options(witherchat-obs-tests PRIVATE /W4 /WX /utf-8 /permissive-)
  add_test(NAME native-worker-support COMMAND witherchat-obs-tests)
endif()
set_target_properties(witherchat-obs PROPERTIES PREFIX "")
install(TARGETS witherchat-obs LIBRARY DESTINATION bin/64bit RUNTIME DESTINATION bin/64bit)
